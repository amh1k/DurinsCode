\begin{array}{l}
\text{Let } G = \langle N, \Sigma, P, S \rangle \text{ be the grammar for Durin's Code, where:} \\
\\
\text{I. Top Level Structures} \\
\langle \text{program} \rangle ::= \langle \text{declaration} \rangle^* \text{ EOF} \\
\langle \text{declaration} \rangle ::= \langle \text{roomDecl} \rangle \mid \langle \text{actionDecl} \rangle \\
\\
\text{II. Room and Entity Declarations} \\
\langle \text{roomDecl} \rangle ::= \text{"room" } \mathbf{STRING} \text{ "{" } \langle \text{roomBody} \rangle^* \text{ "}"} \\
\langle \text{roomBody} \rangle ::= \langle \text{descriptionStmt} \rangle \mid \langle \text{itemDecl} \rangle \mid \langle \text{npcDecl} \rangle \mid \langle \text{exitDecl} \rangle \\
\langle \text{descriptionStmt} \rangle ::= \text{"description" } \mathbf{STRING} \\
\langle \text{itemDecl} \rangle ::= \text{"item" } \mathbf{IDENTIFIER} \text{ "{" } \langle \text{propertyList} \rangle \text{ "}"} \\
\langle \text{npcDecl} \rangle ::= \text{"npc" } \mathbf{IDENTIFIER} \text{ "{" } \langle \text{propertyList} \rangle \text{ "}"} \\
\langle \text{propertyList} \rangle ::= \langle \text{property} \rangle \langle \text{propertyList}' \rangle \\
\langle \text{propertyList}' \rangle ::= \text{"," } \langle \text{property} \rangle \langle \text{propertyList}' \rangle \mid \varepsilon \\
\langle \text{property} \rangle ::= \mathbf{IDENTIFIER} \text{ ":" } \langle \text{literal} \rangle \\
\langle \text{exitDecl} \rangle ::= \text{"exit" } \langle \text{direction} \rangle \text{ } \mathbf{STRING} \\
\langle \text{direction} \rangle ::= \text{"north"} \mid \text{"south"} \mid \text{"east"} \mid \text{"west"} \\
\\
\text{III. Action Logic} \\
\langle \text{actionDecl} \rangle ::= \text{"action" } \mathbf{STRING} \text{ "{" } \langle \text{statementList} \rangle \text{ "}"} \\
\langle \text{statementList} \rangle ::= \langle \text{statement} \rangle^* \\
\langle \text{statement} \rangle ::= \langle \text{ifStmt} \rangle \mid \langle \text{printStmt} \rangle \mid \langle \text{removeStmt} \rangle \mid \langle \text{assignStmt} \rangle \\
\langle \text{ifStmt} \rangle ::= \text{"if" } \langle \text{condition} \rangle \text{ "{" } \langle \text{statementList} \rangle \text{ "}" } \langle \text{ifTail} \rangle \\
\langle \text{ifTail} \rangle ::= \text{"else" "{" } \langle \text{statementList} \rangle \text{ "}"} \mid \varepsilon \\
\langle \text{printStmt} \rangle ::= \text{"print" } \mathbf{STRING} \\
\langle \text{removeStmt} \rangle ::= \text{"remove" } \mathbf{IDENTIFIER} \\
\langle \text{assignStmt} \rangle ::= \text{"player." } \mathbf{IDENTIFIER} \text{ } \langle \text{assignStmtSuffix} \rangle \\
\langle \text{assignStmtSuffix} \rangle ::= \text{"+=" } \mathbf{IDENTIFIER} \mid \text{"=" } \langle \text{literal} \rangle \\
\\
\text{IV. Conditions and Literals} \\
\langle \text{condition} \rangle ::= \langle \text{conditionPrimary} \rangle \langle \text{condition}' \rangle \\
\langle \text{condition}' \rangle ::= \text{"\&\&" } \langle \text{conditionPrimary} \rangle \langle \text{condition}' \rangle \mid \varepsilon \\
\langle \text{conditionPrimary} \rangle ::= \text{"player." } \langle \text{playerCondSuffix} \rangle \mid \text{"current\_room == " } \mathbf{STRING} \\
\langle \text{playerCondSuffix} \rangle ::= \text{"has\_item(" } \mathbf{IDENTIFIER} \text{ ")" } \mid \mathbf{IDENTIFIER} \text{ " == " } \langle \text{literal} \rangle \\
\langle \text{literal} \rangle ::= \mathbf{NUMBER} \mid \mathbf{STRING} \mid \text{"true"} \mid \text{"false"} \mid \mathbf{IDENTIFIER} \\
\end{array}
