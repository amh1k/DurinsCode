\section{Lexer Tests}

\begin{longtable}{p{6.5cm} p{7cm}}
\toprule
\textbf{Test Name} & \textbf{What It Verifies} \\
\midrule
\endhead
\texttt{TokenizesSampleWorldScript} &
  Full world script tokenises correctly end-to-end; checks first 15 tokens including
  \texttt{room}, string, \texttt{\{}, \texttt{description}, \texttt{item}, identifier,
  number, comma, and colon. \\[4pt]

\texttt{RecognizesKeywordsAndPunctuation} &
  All 14 keywords (\texttt{room} through \texttt{current\_room}) plus all punctuation
  tokens are emitted with the correct \texttt{TokenType}. \\[4pt]

\texttt{RecognizesStringAndNumber} &
  A string literal including spaces and a multi-digit integer are tokenised with the
  correct type and lexeme text. \\[4pt]

\texttt{RecognizesCurrentRoomCondition} &
  The sequence \texttt{if current\_room == "bag\_end" \{} produces
  \texttt{TOKEN\_IF}, \texttt{TOKEN\_CURRENT\_ROOM}, \texttt{TOKEN\_EQUAL\_EQUAL},
  \texttt{TOKEN\_STRING}, \texttt{TOKEN\_LEFT\_BRACE} in order. \\[4pt]

\texttt{RecognizesPlayerHasItemAsContextSensitiveAttribute} &
  After \texttt{player.}, the identifier \texttt{has\_item} is emitted as
  \texttt{TOKEN\_PLAYER\_ATTR} (not \texttt{TOKEN\_IDENTIFIER}), confirming the
  context-sensitive dot logic. \\[4pt]

\texttt{RecognizesPlayerAttributeAndPlusEqual} &
  \texttt{player.inventory += the\_ring} produces
  \texttt{TOKEN\_PLAYER}, \texttt{TOKEN\_DOT}, \texttt{TOKEN\_PLAYER\_ATTR},
  \texttt{TOKEN\_PLUS\_EQUAL}, \texttt{TOKEN\_IDENTIFIER}. \\[4pt]

\texttt{ReportsUnexpectedCharacter} &
  The character \texttt{@} produces a \texttt{TOKEN\_ERROR} whose message contains
  ``Unexpected character''. \\[4pt]

\texttt{AcceptsUnderscoreLeadingIdentifiers} &
  \texttt{\_private}, \texttt{\_\_magic}, and \texttt{\_123} are tokenised as
  \texttt{TOKEN\_IDENTIFIER}.
  Also validates that column numbers are tracked correctly
  (\texttt{\_private} at column 1, \texttt{\_\_magic} at column 10). \\[4pt]

\texttt{RejectsFloatingPointNumbers} &
  \texttt{3.14} produces \texttt{TOKEN\_ERROR} with a message containing
  ``Floating-point''. \\[4pt]

\texttt{RecognizesLogicalOrOperator} &
  \texttt{||} produces \texttt{TOKEN\_OR\_OR}. \\[4pt]

\texttt{ReportsColumnNumbers} &
  In the source \texttt{"room\textbackslash n  @\textbackslash naction"}, the
  \texttt{@} error token is reported at line 2, column 3. \\[4pt]

\texttt{HandlesWindowsLineEndings} &
  \texttt{"room\textbackslash r\textbackslash n"test"} tokenises \texttt{room} on
  line 1 and the string on line 2, confirming \texttt{\textbackslash r\textbackslash n}
  is treated as a single newline. \\
\bottomrule
\end{longtable}
